-- 1. Create the database
CREATE DATABASE studentdb;

-- 2. Tell MySQL to use this specific database
USE studentdb;

-- 3. Create the students table
CREATE TABLE students (
    id   INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    cgpa FLOAT        NOT NULL
);

-- 5. To remove a database in MySQL
- Drop a database, all tables and data inside it are deleted instantly
DROP DATABASE studentdb;
- Check if it exists (Best)
DROP DATABASE IF EXISTS studentdb;

-- 6. To remove a table in MySQL
DROP TABLE students;
DROP TABLE IF EXISTS students;

- TRUNCATE vs DELETE
-What it does: Deletes all rows and resets the AUTO_INCREMENT counter back to 1.
-Best for: When you want to start over with a fresh, empty table.
TRUNCATE TABLE students;

-What it does: Deletes all rows but keeps the AUTO_INCREMENT counter where it left off.
-Best for: Specific deletions (e.g., DELETE FROM students WHERE id = 1;).
DELETE FROM students;
